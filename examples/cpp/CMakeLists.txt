cmake_minimum_required(VERSION 3.16)
project(nwt_example LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(nwt-example main.cpp)
target_compile_features(nwt-example PRIVATE cxx_std_20)
find_path(NWT_ZSTD_INCLUDE zstd.h HINTS /opt/homebrew/include /usr/local/include)
find_library(NWT_ZSTD_LIB NAMES zstd HINTS /opt/homebrew/lib /usr/local/lib)
if(NWT_ZSTD_INCLUDE AND NWT_ZSTD_LIB)
  target_compile_definitions(nwt-example PRIVATE NWT_ZSTD)
  target_include_directories(nwt-example PRIVATE ${NWT_ZSTD_INCLUDE})
  target_link_libraries(nwt-example PRIVATE ${NWT_ZSTD_LIB})
else()
  message(STATUS "Zstd not found: building without codec 1 support")
endif()
